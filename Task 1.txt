LAB 8 — MINIMAX (Tic Tac Toe)
================================================================

board = [[' ']*3 for _ in range(3)]

def check_winner(board, player):
    for i in range(3):
        if all(board[i][j] == player for j in range(3)): return True
        if all(board[j][i] == player for j in range(3)): return True
    if all(board[i][i] == player for i in range(3)): return True
    if all(board[i][2-i] == player for i in range(3)): return True
    return False

def is_full(board):
    return all(board[i][j] != ' ' for i in range(3) for j in range(3))

def minimax(board, is_maximizing):
    if check_winner(board, 'X'): return 1    # AI wins  = +1
    if check_winner(board, 'O'): return -1   # Human wins = -1
    if is_full(board): return 0              # Draw = 0

    if is_maximizing:
        best = float('-inf')
        for i in range(3):
            for j in range(3):
                if board[i][j] == ' ':
                    board[i][j] = 'X'
                    best = max(best, minimax(board, False))
                    board[i][j] = ' '   # UNDO move
        return best
    else:
        best = float('inf')
        for i in range(3):
            for j in range(3):
                if board[i][j] == ' ':
                    board[i][j] = 'O'
                    best = min(best, minimax(board, True))
                    board[i][j] = ' '   # UNDO move
        return best

def best_move(board):
    best_score = float('-inf')
    move = None
    for i in range(3):
        for j in range(3):
            if board[i][j] == ' ':
                board[i][j] = 'X'
                score = minimax(board, False)
                board[i][j] = ' '
                if score > best_score:
                    best_score = score
                    move = (i, j)
    return move

def print_board(board):
    for i, row in enumerate(board):
        print(' | '.join(row))
        if i < 2: print('-'*5)

def play():
    print_board(board)
    while True:
        r, c = map(int, input("Enter row col (0-2): ").split())
        if board[r][c] != ' ':
            print("Invalid!"); continue
        board[r][c] = 'O'
        print_board(board)
        if check_winner(board, 'O'): print("You win!"); break
        if is_full(board): print("Draw!"); break
        r, c = best_move(board)
        board[r][c] = 'X'
        print_board(board)
        if check_winner(board, 'X'): print("AI wins!"); break
        if is_full(board): print("Draw!"); break

play()

def alpha_beta(coins, left, right, is_max, alpha, beta):
    if left > right:
        return 0

    if is_max:
        pick_left  = coins[left]  + alpha_beta(coins, left+1, right,   False, alpha, beta)
        pick_right = coins[right] + alpha_beta(coins, left,   right-1, False, alpha, beta)
        best = max(pick_left, pick_right)
        alpha = max(alpha, best)
        if beta <= alpha:
            return best  # PRUNE
        return best
    else:
        pick_left  = alpha_beta(coins, left+1, right,   True, alpha, beta)
        pick_right = alpha_beta(coins, left,   right-1, True, alpha, beta)
        best = min(pick_left, pick_right)
        beta = min(beta, best)
        if beta <= alpha:
            return best  # PRUNE
        return best

coins = [3, 9, 1, 2, 7, 5]
result = alpha_beta(coins, 0, len(coins)-1, True, float('-inf'), float('inf'))
print("Max best score:", result)

import numpy as np

# TRAFFIC SIGNAL (Green=0, Yellow=1, Red=2)
states = ['Green', 'Yellow', 'Red']

# Transition matrix: row=FROM state, col=TO state
# EVERY ROW MUST SUM TO 1.0
transition = np.array([
    [0.6, 0.3, 0.1],   # From Green
    [0.1, 0.2, 0.7],   # From Yellow
    [0.2, 0.1, 0.7],   # From Red
])

def simulate(start_idx, n_steps):
    current = start_idx
    sequence = [states[current]]
    for _ in range(n_steps):
        current = np.random.choice(len(states), p=transition[current])
        sequence.append(states[current])
    return sequence

# Simulate 10 steps from Green
seq = simulate(0, 10)
print(" -> ".join(seq))
print("Red count:", seq.count('Red'))

# P(Red appears >= 4 times in 10 steps)
total = 10000
count = sum(1 for _ in range(total) if simulate(0, 10).count('Red') >= 4)
print(f"P(Red >= 4) = {count/total:.4f}")

# ---- WEATHER (Sunny=0, Cloudy=1, Rainy=2) ----
states_w = ['Sunny', 'Cloudy', 'Rainy']

transition_w = np.array([
    [0.7, 0.2, 0.1],   # From Sunny
    [0.3, 0.4, 0.3],   # From Cloudy
    [0.2, 0.3, 0.5],   # From Rainy
])

def simulate_weather(start_idx, n_steps):
    current = start_idx
    seq = [states_w[current]]
    for _ in range(n_steps):
        current = np.random.choice(len(states_w), p=transition_w[current])
        seq.append(states_w[current])
    return seq

seq_w = simulate_weather(0, 10)  # Start Sunny
print(" -> ".join(seq_w))

# P(at least 3 Rainy days)
count_w = sum(1 for _ in range(10000)
              if simulate_weather(0, 10).count('Rainy') >= 3)
print(f"P(Rainy >= 3) = {count_w/10000:.4f}")

df = pd.read_csv('data.csv')

print("LOOCV Average Accuracy:", np.mean(loo_scores))
